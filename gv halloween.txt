do
    local CONFIG = {
        CHECK_INTERVAL = 1,
        HOUSE_PREFIX = "HalloweenHouse_",
        SPAWN_WAIT = 3,
        CANDY_WAIT = 1.5,
        RESPAWN_WAIT = 180,
    }

    local Players = game:GetService("Players")
    local Workspace = game:GetService("Workspace")
    local VirtualInputManager = game:GetService("VirtualInputManager")
    local LocalPlayer = Players.LocalPlayer

    local character
    local rootPart
    local humanoid
    local isBusy = false
    local savedRootCFrame = nil
    local unusedState = nil
    local savedCameraType = nil
    local savedCameraSubject = nil

    local function getCharacterParts(newCharacter)
        character = newCharacter
        rootPart = character:WaitForChild("HumanoidRootPart")
        humanoid = character:WaitForChild("Humanoid")
    end

    getCharacterParts(LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait())
    LocalPlayer.CharacterAdded:Connect(function(newCharacter)
        getCharacterParts(newCharacter)
    end)

    local function findHouseModel()
        for _, child in ipairs(Workspace:GetChildren()) do
            if child:IsA("Model") and string.sub(child.Name, 1, #CONFIG.HOUSE_PREFIX) == CONFIG.HOUSE_PREFIX then
                return child
            end
        end
        return nil
    end

    local function findDoorPrompt(house)
        if not house then
            return nil
        end
        local firstPrompt = nil

        for _, descendant in ipairs(house:GetDescendants()) do
            if descendant:IsA("ProximityPrompt") then
                if not firstPrompt then
                    firstPrompt = descendant
                end
                local ancestor = descendant.Parent
                while ancestor and (ancestor ~= house) do
                    if ancestor:IsA("Model") and string.find(string.lower(ancestor.Name), "plane") then
                        return descendant
                    end
                    ancestor = ancestor.Parent
                end
            end
        end
        return firstPrompt
    end

    local function getPromptWorldPosition(prompt)
        if not prompt then
            return nil
        end
        local parent = prompt.Parent
        if parent:IsA("Attachment") then
            return parent.WorldPosition
        elseif parent:IsA("BasePart") then
            return parent.Position
        elseif parent:IsA("Model") then
            return parent:GetPivot().Position
        end
        return nil
    end

    local function saveCamera()
        local camera = Workspace.CurrentCamera
        if not camera then
            return
        end
        savedCameraType = camera.CameraType
        savedCameraSubject = camera.CameraSubject
    end

    local function pointCameraAt(target)
        local camera = Workspace.CurrentCamera
        if not camera or not rootPart then
            return
        end
        camera.CameraType = Enum.CameraType.Scriptable
        camera.CFrame = CFrame.lookAt(camera.CFrame.Position, target)
    end

    local function restoreCamera()
        local camera = Workspace.CurrentCamera
        if not camera then
            return
        end
        camera.CameraType = savedCameraType or Enum.CameraType.Custom
        camera.CameraSubject = humanoid or savedCameraSubject
        savedCameraType = nil
        savedCameraSubject = nil
    end

    local function teleportTo(position)
        if not rootPart then
            return false
        end
        rootPart.Anchored = false
        rootPart.CFrame = CFrame.new(position + Vector3.new(0, 3, 0))
        rootPart.AssemblyLinearVelocity = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
        return true
    end

    local function returnToSavedPosition()
        if savedRootCFrame and rootPart then
            rootPart.CFrame = savedRootCFrame
            rootPart.AssemblyLinearVelocity = Vector3.zero
            rootPart.AssemblyAngularVelocity = Vector3.zero
        end
        savedRootCFrame = nil
        restoreCamera()
    end

    local function pressInteractKey()
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
        task.wait(0.5)
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
    end

    local function tryActivatePrompt(prompt)
        if not prompt or not prompt.Parent then
            return false
        end
        local promptPosition = getPromptWorldPosition(prompt)
        if not promptPosition then
            return false
        end

        saveCamera()
        local promptParent = prompt.Parent
        local facing = Vector3.new(0, 0, 1)

        if promptParent:IsA("BasePart") then
            facing = promptParent.CFrame.LookVector
        elseif promptParent:IsA("Attachment") and promptParent.Parent:IsA("BasePart") then
            facing = promptParent.Parent.CFrame.LookVector
        end

        local standPosition = promptPosition + (facing * 3)
        if not teleportTo(standPosition) then
            restoreCamera()
            return false
        end

        task.wait(0.3)
        pointCameraAt(promptPosition)
        task.wait(0.3)

        if not prompt.Parent or not prompt.Enabled then
            return false
        end
        local distance = (rootPart.Position - promptPosition).Magnitude
        if distance > prompt.MaxActivationDistance then
            return false
        end

        pressInteractKey()
        return true
    end

    local function mainLoop()
        while true do
            if isBusy then
                task.wait(CONFIG.CHECK_INTERVAL)
                continue
            end

            local house = findHouseModel()
            if not house then
                task.wait(CONFIG.CHECK_INTERVAL)
                continue
            end

            isBusy = true
            task.wait(CONFIG.SPAWN_WAIT)

            if house.Parent and rootPart then
                savedRootCFrame = rootPart.CFrame
                local doorPrompt = findDoorPrompt(house)
                if doorPrompt then
                    local activated = tryActivatePrompt(doorPrompt)
                    if activated then
                        task.wait(CONFIG.CANDY_WAIT)
                    end
                end
                returnToSavedPosition()
            end

            isBusy = false
            task.wait(CONFIG.RESPAWN_WAIT)
        end
    end

    task.spawn(mainLoop)
end

do
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer
    if not LocalPlayer then
        warn("LocalPlayer nicht gefunden")
        return
    end

    local playerGui = LocalPlayer:WaitForChild("PlayerGui")

    local OWNER_NAME = "nigger"
    local SHOW_DISTANCE = 50

    local ownerTag
    local trackedCharacter

    local function removeOwnerTag()
        if ownerTag then
            ownerTag:Destroy()
            ownerTag = nil
        end
        trackedCharacter = nil
    end

    local function updateOwnerTag()
        local ownerPlayer = Players:FindFirstChild(OWNER_NAME)
        if not ownerPlayer then
            removeOwnerTag()
            return
        end

        local ownerCharacter = ownerPlayer.Character
        local ownerHead = ownerCharacter and ownerCharacter:FindFirstChild("Head")
        if not ownerHead then
            removeOwnerTag()
            return
        end

        if (trackedCharacter ~= ownerCharacter) or not ownerTag or not ownerTag.Parent then
            removeOwnerTag()
            trackedCharacter = ownerCharacter

            ownerTag = Instance.new("BillboardGui")
            ownerTag.Name = "CandyFarmOwnerTag"
            ownerTag.Adornee = ownerHead
            ownerTag.Size = UDim2.fromOffset(300, 35)
            ownerTag.StudsOffset = Vector3.new(0, 3.5, 0)
            ownerTag.AlwaysOnTop = true
            ownerTag.Enabled = false
            ownerTag.Parent = playerGui

            local tagLabel = Instance.new("TextLabel")
            tagLabel.Size = UDim2.fromScale(1, 1)
            tagLabel.BackgroundTransparency = 1
            tagLabel.Text = "Candy Farm Owner"
            tagLabel.TextColor3 = Color3.new(1, 1, 1)
            tagLabel.TextStrokeTransparency = 0
            tagLabel.Font = Enum.Font.BuilderSansBold
            tagLabel.TextSize = 24
            tagLabel.ZIndex = 10
            tagLabel.Parent = ownerTag
        end

        local myCharacter = LocalPlayer.Character
        local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
        local ownerRoot = ownerCharacter:FindFirstChild("HumanoidRootPart")
        if not myRoot or not ownerRoot then
            ownerTag.Enabled = false
            return
        end

        ownerTag.Enabled = (myRoot.Position - ownerRoot.Position).Magnitude <= SHOW_DISTANCE
    end

    task.spawn(function()
        while task.wait(0.25) do
            local ok, err = pcall(updateOwnerTag)
            if not ok then
                warn("Owner Tag:", err)
            end
        end
    end)
end
