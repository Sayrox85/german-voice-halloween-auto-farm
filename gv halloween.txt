local SPAWN_ATTR    = "HalloweenHouseSpawnId"
local HOUSE_PREFIX  = "HalloweenHouse"
local RETURN_AFTER  = true
local STREAM_WAIT   = 0.4
local STAND_WAIT    = 0.08
local MAX_LOG       = 100

local state = {
    log         = {},
    processed   = {},
    knownHouses = {},
    farmed      = 0,
    enabled     = true,
    conns       = {},
}
_G.HalloweenFarm = state

local function push(msg)
    table.insert(state.log, string.format("[%.1f] %s", os.clock(), msg))
    if #state.log > MAX_LOG then table.remove(state.log, 1) end
end

local function getHRP()
    local char = LP.Character or LP.CharacterAdded:Wait()
    return char:FindFirstChild("HumanoidRootPart") or char:WaitForChild("HumanoidRootPart", 5)
end

local function isHouse(inst)
    if not inst:IsA("Model") then return false end
    if inst:GetAttribute(SPAWN_ATTR) then return true end
    if inst.Name:sub(1, #HOUSE_PREFIX) == HOUSE_PREFIX then return true end
    return false
end

local function findDoorbell(house)
    local prompt, click, named
    for _, d in ipairs(house:GetDescendants()) do
        local n = d.Name:lower()
        if (d:IsA("ProximityPrompt") or d:IsA("ClickDetector") or d:IsA("BasePart"))
           and (n:find("doorbell") or n:find("bell") or n:find("button") or n:find("ring") or n:find("press"))
        then
            named = named or d
        elseif d:IsA("ProximityPrompt") then
            prompt = prompt or d
        elseif d:IsA("ClickDetector") then
            click = click or d
        end
    end

    if not named then
        for _, d in ipairs(house:GetDescendants()) do
            if d:IsA("ProximityPrompt") then
                local meta = (d.ObjectText or "")..(d.ActionText or "")
                if meta:lower():find("doorbell") or meta:lower():find("ring") then
                    named = d
                    break
                end
            end
        end
    end

    if named then
        if named:IsA("ProximityPrompt") then return "prompt", named, named.Parent end
        if named:IsA("ClickDetector")   then return "click",  named, named.Parent end
        for _, c in ipairs(named:GetDescendants()) do
            if c:IsA("ProximityPrompt") then return "prompt", c, named end
            if c:IsA("ClickDetector")   then return "click",  c, named end
        end
        return "touch", named, named
    end
    if prompt then return "prompt", prompt, prompt.Parent end
    if click  then return "click",  click,  click.Parent  end
    return nil
end

local function fire(kind, target, anchor)
    local hrp = getHRP()
    if not hrp or not anchor or not anchor:IsA("BasePart") then
        push("no HRP or bad anchor; abort")
        return false
    end
    local origin      = hrp.CFrame
    local wasAnchored = hrp.Anchored

    hrp.Anchored = true
    hrp.AssemblyLinearVelocity  = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.CFrame = anchor.CFrame * CFrame.new(0, 0, -2)

    local ok = false
    if kind == "prompt" then
        ok = select(1, pcall(fireproximityprompt, target, 1))
        task.wait(STAND_WAIT)
    elseif kind == "click" then
        ok = select(1, pcall(fireclickdetector, target, 0, "MouseClick"))
        task.wait(STAND_WAIT)
    elseif kind == "touch" then
        ok = select(1, pcall(firetouchinterest, hrp, target, 0))
        task.wait(0.05)
        pcall(firetouchinterest, hrp, target, 1)
    end

    if RETURN_AFTER then
        hrp.CFrame = origin
        hrp.AssemblyLinearVelocity  = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
    hrp.Anchored = wasAnchored
    return ok
end

local function tryFarm(house)
    if not state.enabled then return end
    if not house:IsA("Model") then return end

    for _ = 1, 20 do
        if house:GetAttribute(SPAWN_ATTR) then break end
        task.wait(0.1)
        if not house.Parent then return end
    end
    local id = house:GetAttribute(SPAWN_ATTR)
    if not id or state.processed[id] then return end
    state.processed[id] = true

    push("house "..tostring(id):sub(1,8).." scanning")
    task.wait(STREAM_WAIT)

    if not state.enabled then
        state.processed[id] = nil
        return
    end
    if not house.Parent then
        state.processed[id] = nil
        return
    end

    local kind, target, anchor = findDoorbell(house)
    if not kind then
        push("no interactable found inside house")
        return
    end
    local ok = fire(kind, target, anchor)
    if ok then
        state.farmed = state.farmed + 1
        push("rang via "..kind)
    else
        push("fire failed on "..kind)
    end
end

local function registerHouse(inst)
    if state.knownHouses[inst] then return end
    state.knownHouses[inst] = true
    if state.enabled then task.spawn(tryFarm, inst) end
end

local function onChildAdded(d)
    if isHouse(d) then registerHouse(d) end
end

local function onChildRemoved(d)
    if state.knownHouses[d] then
        state.knownHouses[d] = nil
        local id = d:GetAttribute(SPAWN_ATTR)
        if id then
            task.delay(8, function() state.processed[id] = nil end)
        end
    end
end

table.insert(state.conns, Workspace.ChildAdded:Connect(onChildAdded))
table.insert(state.conns, Workspace.ChildRemoved:Connect(onChildRemoved))

for _, d in ipairs(Workspace:GetChildren()) do
    if isHouse(d) then registerHouse(d) end
end

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 6)

    local s = Instance.new("UIStroke")

    s.Thickness = thick or 1
    s.Parent = parent
    return s
end

local statFarmed = mkStat(70, "Farmed")
local statHouses = mkStat(88, "Houses")
local statLast   = mkStat(106, "Last")

local function applyEnabled(on)
    state.enabled = on and true or false
    if state.enabled then
        toggle.Text = "ON"
        toggle.BackgroundColor3 = GREEN
        push("enabled")
        for house in pairs(state.knownHouses) do
            if house.Parent then task.spawn(tryFarm, house) end
        end
    else
        toggle.Text = "OFF"
        toggle.BackgroundColor3 = RED
        push("disabled")
    end
end

function state.setEnabled(v) applyEnabled(v) end
toggle.MouseButton1Click:Connect(function() applyEnabled(not state.enabled) end)

do
    local dragging, dragStart, startPos
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
           or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos  = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

function state.stop()
    if state.stopped then return end
    state.stopped = true
    state.enabled = false
    for _, c in ipairs(state.conns) do pcall(function() c:Disconnect() end) end
    state.conns = {}
    pcall(function()
        local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if hrp then hrp.Anchored = false end
    end)
    if gui and gui.Parent then gui:Destroy() end
    push("stopped")
    _G.HalloweenFarm = nil
end
closeBtn.MouseButton1Click:Connect(function() state.stop() end)

task.spawn(function()
    while gui.Parent do
        local houses = 0
        for h in pairs(state.knownHouses) do
            if h.Parent then houses = houses + 1 end
        end
        statFarmed.Text = tostring(state.farmed)
        statHouses.Text = tostring(houses)
        local lastLine = state.log[#state.log]
        if lastLine then
            statLast.Text = lastLine:gsub("^%[[^%]]+%]%s*", "")
        end
        task.wait(1)
    end
end)

push("auto-farm armed; waiting for houses")
